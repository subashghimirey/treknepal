**Pseudocode:**

```
ALGORITHM: User_Registration
INPUT: username, email, password, display_name
OUTPUT: auth_token, user_profile

BEGIN
    // Step 1: Validate Input
    IF username is EMPTY OR password is EMPTY:
        RETURN error("Required fields missing")
    
    IF LENGTH(password) < 8:
        RETURN error("Password must be at least 8 characters")
    
    IF email format is INVALID:
        RETURN error("Invalid email format")
    
    // Step 2: Check Uniqueness
    IF User.exists(username):
        RETURN error("Username already taken")
    
    IF User.exists(email):
        RETURN error("Email already registered")
    
    // Step 3: Hash Password
    hashed_password = bcrypt.hash(password, salt_rounds=12)
    
    // Step 4: Create User
    user = User.create({
        username: username,
        email: email,
        password: hashed_password,
        first_name: display_name,
        is_active: TRUE
    })
    
    // Step 5: Create UserProfile
    profile = UserProfile.create({
        user_id: user.id,
        display_name: display_name,
        role: 'user',
        interests: [],
        is_active: TRUE
    })
    
    // Step 6: Generate Token
    token = Token.create(user)
    // Token format: 40 random hex chars
    // Example: "9944b09199c62bcf9418ad846dd0e4bbdfc6ee4b"
    
    // Step 7: Return Response
    RETURN {
        token: token.key,
        user: {
            id: profile.id,
            username: user.username,
            email: user.email,
            display_name: profile.display_name,
            role: profile.role,
            photo_url: profile.photo_url,
            interests: profile.interests
        }
    }
END
```

**Pseudocode:**

```
ALGORITHM: User_Login
INPUT: username, password
OUTPUT: auth_token, user_profile

BEGIN
    // Step 1: Authenticate User
    user = authenticate(username, password)
    
    IF user is NULL:
        RETURN error("Invalid credentials", status=401)
    
    IF user.is_active is FALSE:
        RETURN error("Account is disabled", status=403)
    
    // Step 2: Get or Create Token
    TRY:
        token = Token.get(user_id=user.id)
    CATCH TokenDoesNotExist:
        token = Token.create(user_id=user.id)
    
    // Step 3: Get UserProfile
    TRY:
        profile = UserProfile.get(user_id=user.id)
    CATCH UserProfileDoesNotExist:
        // Create profile if doesn't exist
        profile = UserProfile.create({
            user_id: user.id,
            display_name: user.username,
            role: 'user'
        })
    
    // Step 4: Update Last Login
    user.last_login = CURRENT_TIMESTAMP()
    user.save()
    
    // Step 5: Return Response
    RETURN {
        token: token.key,
        user: serialize(profile),
        message: "Login successful"
    }
END
```

**Pseudocode:**

```
ALGORITHM: Password_Reset_Request_OTP
INPUT: email
OUTPUT: success_message

BEGIN
    // Step 1: Find User
    TRY:
        user = User.get(email=email)
    CATCH UserDoesNotExist:
        RETURN error("No user found with this email", 404)
    
    // Step 2: Generate OTP
    otp = RANDOM_NUMBER(6_digits)  // e.g., "482751"
    
    // Step 3: Save OTP
    PasswordResetOTP.create({
        user_id: user.id,
        otp: otp,
        created_at: CURRENT_TIMESTAMP(),
        is_used: FALSE,
        is_verified: FALSE
    })
    
    // Step 4: Send Email
    email_body = "Your password reset code is: " + otp + 
                 "\nValid for 10 minutes."
    
    send_email(
        to: email,
        subject: "Password Reset Code",
        body: email_body
    )
    
    RETURN {
        success: TRUE,
        message: "Password reset code sent to your email"
    }
END


ALGORITHM: Verify_OTP
INPUT: otp
OUTPUT: verification_status

BEGIN
    // Step 1: Find OTP
    TRY:
        otp_obj = PasswordResetOTP.filter(
            otp=otp,
            is_used=FALSE
        ).latest('created_at')
    CATCH DoesNotExist:
        RETURN error("Invalid OTP", 400)
    
    // Step 2: Check Expiry
    time_elapsed = CURRENT_TIMESTAMP() - otp_obj.created_at
    
    IF time_elapsed > 10_MINUTES:
        RETURN error("OTP has expired", 400)
    
    // Step 3: Mark as Verified
    otp_obj.is_verified = TRUE
    otp_obj.save()
    
    RETURN {
        success: TRUE,
        verified: TRUE,
        message: "OTP verified successfully"
    }
END


ALGORITHM: Set_New_Password
INPUT: otp, new_password
OUTPUT: success_message

BEGIN
    // Step 1: Validate OTP
    TRY:
        otp_obj = PasswordResetOTP.filter(
            otp=otp,
            is_verified=TRUE,
            is_used=FALSE
        ).latest('created_at')
    CATCH DoesNotExist:
        RETURN error("Invalid or unverified OTP", 400)
    
    // Step 2: Check Expiry Again
    time_elapsed = CURRENT_TIMESTAMP() - otp_obj.created_at
    IF time_elapsed > 10_MINUTES:
        RETURN error("OTP has expired", 400)
    
    // Step 3: Hash New Password
    hashed_password = bcrypt.hash(new_password, salt_rounds=12)
    
    // Step 4: Update User Password
    user = otp_obj.user
    user.password = hashed_password
    user.save()
    
    // Step 5: Mark OTP as Used
    otp_obj.is_used = TRUE
    otp_obj.save()
    
    RETURN {
        success: TRUE,
        message: "Password reset successful"
    }
END
```

### **1.5 Security Features**

#### **A. Password Hashing**

```python
# Django uses PBKDF2 with SHA256 by default
# Format: <algorithm>$<iterations>$<salt>$<hash>

Example Stored Password:
"pbkdf2_sha256$260000$randomsalt$hashedpassword..."

Breakdown:
├─ Algorithm: pbkdf2_sha256
├─ Iterations: 260,000 (as of Django 3.2+)
├─ Salt: Random 22-character string
└─ Hash: 44-character base64 encoded hash

Security Benefits:
✓ Salted (prevents rainbow table attacks)
✓ 260,000 iterations (computationally expensive)
✓ SHA256 hashing algorithm
✓ Cannot be reversed
```

#### **B. Token Authentication**

```python
# Token Format
Token Key: "9944b09199c62bcf9418ad846dd0e4bbdfc6ee4b"
Length: 40 hexadecimal characters
Entropy: 160 bits (2^160 possible combinations)

# HTTP Header
Authorization: Token 9944b09199c62bcf9418ad846dd0e4bbdfc6ee4b

Security Features:
✓ Cryptographically secure random generation
✓ Stored in database (can be invalidated)
✓ No expiration (logout to invalidate)
✓ One token per user
```

#### **C. OTP Security**

```python
# OTP Specifications
Format: 6-digit numeric code
Range: 100000 - 999999
Validity: 10 minutes
One-time use: Marked as used after consumption

Security Features:
✓ Time-limited (10 minutes)
✓ Single use only
✓ Random generation
✓ Email delivery (out-of-band verification)
✓ Cannot be reused
```

### **2.5 Post Creation Pseudocode**

```
ALGORITHM: Create_Post
INPUT: user_token, trek_id, content, images[], location
OUTPUT: created_post

BEGIN
    // Step 1: Authenticate User
    user = authenticate_token(user_token)
    IF user is NULL:
        RETURN error("Unauthorized", 401)
    
    // Step 2: Validate Trek
    TRY:
        trek = Trek.get(id=trek_id)
    CATCH TrekDoesNotExist:
        RETURN error("Trek not found", 404)
    
    // Step 3: Validate Content
    IF content is EMPTY:
        RETURN error("Content is required", 400)
    
    IF LENGTH(content) > 5000:
        RETURN error("Content too long (max 5000 chars)", 400)
    
    // Step 4: Upload Images to Cloudinary
    uploaded_images = []
    
    FOR each image IN images:
        // Validate image
        IF NOT is_valid_image(image):
            RETURN error("Invalid image format", 400)
        
        IF image.size > 10_MB:
            RETURN error("Image too large (max 10MB)", 400)
        
        // Upload to Cloudinary
        TRY:
            result = cloudinary.uploader.upload(
                image,
                folder="trek_posts",
                transformation=[
                    {width: 1200, height: 1200, crop: "limit"},
                    {quality: "auto"},
                    {fetch_format: "auto"}
                ]
            )
            uploaded_images.APPEND(result.secure_url)
        CATCH CloudinaryError as e:
            RETURN error("Image upload failed: " + e.message, 500)
    
    // Step 5: Validate Location (Optional)
    IF location is NOT NULL:
        IF location.latitude < -90 OR location.latitude > 90:
            RETURN error("Invalid latitude", 400)
        
        IF location.longitude < -180 OR location.longitude > 180:
            RETURN error("Invalid longitude", 400)
    
    // Step 6: Create Post
    post = Post.create({
        user_id: user.profile.id,
        trek_id: trek.id,
        content: content,
        images: uploaded_images,  // Array of Cloudinary URLs
        location: location,
        likes_count: 0,
        comments_count: 0,
        status: 'active',
        created_at: CURRENT_TIMESTAMP()
    })
    
    // Step 7: Serialize Response
    response = {
        success: TRUE,
        message: "Post created successfully",
        post: {
            id: post.id,
            content: post.content,
            images: post.images,
            location: post.location,
            likes_count: post.likes_count,
            comments_count: post.comments_count,
            created_at: post.created_at,
            user: {
                id: user.profile.id,
                display_name: user.profile.display_name,
                photo_url: user.profile.photo_url
            },
            trek: {
                id: trek.id,
                name: trek.name,
                region: trek.region,
                difficulty: trek.difficulty
            }
        }
    }
    
    RETURN response (status=201)
END
```

**Pseudocode:**

```
ALGORITHM: Toggle_Post_Like
INPUT: user_token, post_id
OUTPUT: like_status

BEGIN
    // Step 1: Authenticate
    user = authenticate_token(user_token)
    IF user is NULL:
        RETURN error("Unauthorized", 401)
    
    // Step 2: Get Post
    TRY:
        post = Post.get(id=post_id)
    CATCH PostDoesNotExist:
        RETURN error("Post not found", 404)
    
    // Step 3: Check Existing Like
    TRY:
        like = Like.get(post_id=post_id, user_id=user.profile.id)
        
        // Like exists, so unlike
        like.delete()
        
        // Decrement likes count
        post.likes_count = post.likes_count - 1
        post.save()
        
        RETURN {status: "unliked", likes_count: post.likes_count}
        
    CATCH LikeDoesNotExist:
        // Like doesn't exist, so create it
        Like.create(post_id=post_id, user_id=user.profile.id)
        
        // Increment likes count
        post.likes_count = post.likes_count + 1
        post.save()
        
        RETURN {status: "liked", likes_count: post.likes_count}
END
```
